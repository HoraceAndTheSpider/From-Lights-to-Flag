Wait Vbl : Auto View Off : Wait Vbl 

' 10 = LOCATIONS.DATA
' 11 = SURFACES.MAP
' 12 = BACKGROUND.IFF  
' 13 = FOREGROUND.IFF

' LOCATION DATA
 Erase 10

 If Exist(A$) : Wload A$,10 : Bank Name 10,"TrackDta" : End If

 End If

   Spack FGSCR To 13
   Bank Name 13,"ForgdPac"
 End If

 If Exist(A$) and Length(13)=0
   Wload A$,13
   Unpack 13 To FGSCR : Screen Hide FGSCR
   Bank Name 13,"ForgdPac"
 End If 

Screen Close FGSCR


Gosub TRACK_DATA_READ


Return 


_FILE_HEALTH:

' track folder OK
If Exist(TRACKPATH$)=False or TRACKPATH$=""
 TRACKPATH$=""
 MSG$="No tracks folder found" : REFRESH=True : ACTION=0 : Return 
End If 

If Right$(TRACKPATH$,1)<>"/" Then TRACKPATH$=TRACKPATH$+"/"

If Exist(TRACKPATH$+TRACKFILE$)=False
 TRACKFILE$=""
End If 

' loaded a track?
If TRACKFILE$=""
 MSG$="No track loaded" : REFRESH=True : ACTION=0 : Return 
End If 


' files in folder

LOCDATA=Exist(TRACKPATH$+TRACKFILE$+"/locations.data")
SUFDATA=Exist(TRACKPATH$+TRACKFILE$+"/surfaces.map")
BKGDATA=Exist(TRACKPATH$+TRACKFILE$+"/background.iff") or Exist(TRACKPATH$+"/background.pac")
FGDDATA=Exist(TRACKPATH$+TRACKFILE$+"/foreground.iff") or Exist(TRACKPATH$+"/foreground.pac")


Return

SMALLTYPE:
   If MYX=-1 Then MYX=160-(Len(MYTEXT$)*3)
   If Len(MYTEXT$)<1 Then Goto SKIPPY
   DIGIT=0
   Repeat 
     CHAR=Asc(Left$(MYTEXT$,1))
     CHAR=CHAR-31
       If Len(MYTEXT$)>1
         MYTEXT$=Right$(MYTEXT$,Len(MYTEXT$)-1)
       Else 
         MYTEXT$=""
       End If 

Rem >> Screen 0

Rem >>>     If CHAR<>1 Then Paste Bob MYX+(DIGIT*6),MYY,CHAR 

      If CHAR<>1 Then Blitter Wait : Paste Bob MYX+(DIGIT*6),MYY,CHAR-(ACTIVE*160)
      If MYDELAY>0 and CHAR<>1 Then Wait MYDELAY

Rem >>      View 
     DESTX=DESTX+8 : Inc DIGIT
   Until Len(MYTEXT$)<1
SKIPPY:
   MYX=0 : MYY=0 : MYTEXT$="" : MYDELAY=False

Return 

LARGETYPE:
   If MYX=-1 Then MYX=160-(Len(MYTEXT$)*4)
   If Len(MYTEXT$)<1 Then Goto SKIPPY2
   DIGIT=0 : MYTEXT$=Upper$(MYTEXT$)
   Repeat 
     CHAR=Asc(Left$(MYTEXT$,1))
     CHAR=CHAR+256+33
       If Len(MYTEXT$)>1
         MYTEXT$=Right$(MYTEXT$,Len(MYTEXT$)-1)
       Else 
         MYTEXT$=""
       End If 
Rem >>     Screen 0
     If CHAR<>1 Then Blitter Wait : Paste Bob MYX+(DIGIT*8),MYY,CHAR-(ACTIVE*160)

 Rem >>    Paste Bob MYX+(DIGIT*8),MYY,CHAR 
      If MYDELAY>0 Then Wait MYDELAY
Rem >>      View 
     DESTX=DESTX+10 : Inc DIGIT
   Until Len(MYTEXT$)<1
 SKIPPY2:
   MYX=0 : MYY=0 : MYTEXT$="" : MYDELAY=False
Return
