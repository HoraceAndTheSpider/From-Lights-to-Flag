   If RACECOUNT=0 : Pop Proc : End If

    If PLCOM(PLYA)>0
       Dec PLCOM(PLYA)
    End If 

  ' limit us to one player comms per turn  
    If ANYCOMM=True : Pop Proc : End If 

  PICKED=True
  COUNT=0

 ' randomiser 
  If Rnd(4)<2 : PICKED=False : End If

  If PLCOM(PLYA)>0 : PICKED=False : End If

           If SCENARIO<3 : SCENARIO=1

      Else If SCENARIO<7 : SCENARIO=3
      Else If SCENARIO<9 : SCENARIO=5

      Rem SCENARIO=8 


      '
      ' pick a 'rival' for scenario 7 (team mate) and scenario 8 (rival) 
      If SCENARIO=8

           Repeat 
              RIVAL=Rnd(3)
           Until RIVAL<>PLYA

           CHAR$=PLYR_NAMES$(RIVAL)

      Else If SCENARIO=7

        If Odd(RIVAL) : RIVAL=RIVAL-1 : Else : RIVAL=RIVAL+1 : End If

       End If 


     _COMMS[PLYA,SCENARIO,CHAR_GRAF,CHAR$,_TEAM$(PLYA),PLYR_NAMES$(PLYA)]


  Rem Next

Procedure _SUPER_RAINBOW[_BANK,_INDEX,_POSITION,_RAIN_NUMBER]
If Prg State=-1 : FIDDLE=-1 : Else FIDDLE=0 : End If 
   ID$=""
   For IDCHAR=0 To 7
      ID$=ID$+Chr$(Peek(Start(_BANK)-8+IDCHAR))
   Next IDCHAR
   If ID$="Rainbow "
      Set Rainbow _RAIN_NUMBER,_INDEX,(Length(_BANK)/2)+FIDDLE,"","",""

Procedure _WAIT_ACTIVE_PLAYER[TIMELIMIT,PLYA]

COUNT=-1
Repeat 
   Wait Vbl 
   For ZZZ=0 To 3

      INPPY=0

        If PLMODEN(ZZZ)=1 : INPPY=Joy(1)
   Else If PLMODEN(ZZZ)=2 : INPPY=Joy(0)
   Else If PLMODEN(ZZZ)=3 : INPPY=Ext Joy(0)-(Pfire(0)*16)
   Else If PLMODEN(ZZZ)=4 : INPPY=Ext Joy(1)-(Pfire(1)*16)
   Else 
      If PLMODEN(ZZZ)=5 and Key State($4C)=True : Bset 0,INPPY : End If 
      If PLMODEN(ZZZ)=5 and Key State($4D)=True : Bset 1,INPPY : End If 
      If PLMODEN(ZZZ)=5 and Key State($4F)=True : Bset 2,INPPY : End If

      If PLYA<>-1 and PLYA<>ZZZ : INPPY=0 : End If 
      If INPPY>15 : Exit : End If 
   Next

Until INPPY>15 or Key State($40)=True or COUNT>=TIMELIMIT
